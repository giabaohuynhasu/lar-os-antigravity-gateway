V. Heartbeat nên là FILE, không phải HTTP

Mình thích file hơn /health cho watchdog nội bộ.

C:\...\lar_os_heartbeat.json

Gateway ghi atomically:

Python
Chạy
import json
import os
import tempfile
import time


def write_heartbeat(path, pid, state):

    payload = {
        "ts": time.time(),
        "pid": pid,
        "state": state,
    }

    directory = os.path.dirname(path)

    fd, tmp = tempfile.mkstemp(
        prefix=".heartbeat.",
        dir=directory,
        text=True,
    )

    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(payload, f)
            f.flush()
            os.fsync(f.fileno())

        os.replace(tmp, path)

    except Exception:
        try:
            os.unlink(tmp)
        except OSError:
            pass

Atomic replace tránh watcher đọc phải file JSON đang ghi dở.

VI. Watchdog logic

Một external watchdog chỉ cần logic kiểu này:

Python
Chạy
import json
import os
import time


HEARTBEAT_MAX_AGE = 30.0


def gateway_alive(pid):
    try:
        os.kill(pid, 0)
        return True
    except OSError:
        return False


def read_heartbeat(path):

    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return None


def nuclear_state(path):

    hb = read_heartbeat(path)

    if hb is None:
        return "HEARTBEAT_MISSING"

    age = time.time() - hb["ts"]

    if age > HEARTBEAT_MAX_AGE:
        return "EVENT_LOOP_HANG"

    if not gateway_alive(hb["pid"]):
        return "PROCESS_DEAD"

    return "HEALTHY"

Watcher chạy:

sleep 10s
↓
read heartbeat
↓
check PID
↓
evaluate
↓
sleep

CPU idle ~0%.

def excepthook(exc_type, exc, tb):
    try:
        import traceback

        with open(
            CRASH_LOG,
            "a",
            encoding="utf-8",
        ) as f:
            traceback.print_exception(
                exc_type,
                exc,
                tb,
                file=f,
            )
    finally:
        sys.__excepthook__(
            exc_type,
            exc,
            tb,
        )


sys.excepthook = excepthook
